   **Pipeline A adapter** (per RESEARCH Pitfall 2 — wrap in v1.1's [-1,1] rescale so generator output range matches):
   ```python
   raw = load_and_preprocess(csv_path)
   od = raw["OD"]                                           # (778,) float32
   scaled01, od_min, od_max = forward_minmax_od(od)         # [0, 1]
   # Map [0,1] -> [-1,1] to match training-loop generator output range:
   scaled_pm1 = 2.0 * scaled01 - 1.0
   windowed = rolling_window(scaled_pm1, WINDOW_LENGTH, 2)  # (M, 10)
   ds = TensorDataset(windowed.float())
   ```

   **Pipeline B adapter** (per D-09.1-01, CD-5 → real per-window OD₀):
   ```python
   raw = load_and_preprocess(csv_path)
   od = raw["OD"]
   r_norm, mu, sigma = forward_logreturns(od)               # (777,) standardized log-returns
   # Map standardized log-returns into [-1,1] using their own min/max:
   r_min, r_max = torch.min(r_norm), torch.max(r_norm)
   r_pm1 = 2.0 * (r_norm - r_min) / (r_max - r_min) - 1.0
   windowed = rolling_window(r_pm1, WINDOW_LENGTH, 2)       # (M, 10)
   # Cache real per-window OD0 values, aligned with windowed_data:
   # window i starts at log-return index 2*i, which corresponds to OD index 2*i
   # (log-return r_t = ln(OD_{t+1}/OD_t), so r at index k uses OD[k]).
   n_lr = r_norm.shape[0]
   start_indices = list(range(0, n_lr - WINDOW_LENGTH + 1, 2))  # stride 2
   od_starts = torch.stack([od[i] for i in start_indices])      # (M,) float32
   ds = TensorDataset(windowed.float())
   dl = DataLoader(ds, batch_size=BATCH_SIZE, shuffle=False, drop_last=True)
   inverse_kwargs = {
       "r_min": r_min.item(), "r_max": r_max.item(),
       "mu": mu.item(), "sigma": sigma.item(),
       "od_starts": od_starts.cpu().numpy(),   # serialize as list for YAML
   }
   ```

   ```python
       all_samples = []
       remaining = n_synth
       # Use np.random.default_rng(seed) per D-09.1-18 for noise generation:

       while remaining > 0:
   ```

5. **Pipeline C parity gate (D-09.1-12)**:
   ```python
   import json, torch
   from revision.core.data import load_and_preprocess
   from revision.core.eval import compute_emd
   d = load_and_preprocess("./data.csv")
   real_log_delta = d["log_delta"].numpy()              # length 777, real log-returns
   # Reconstruct Pipeline C synthetic log-returns from samples_pm1 by un-doing
   # the [-1,1] scaling and the inverse-Lambert transform, the way the training

   samples_C = np.load("revision/results/transform_ablation/runs/C/42/samples.npy")
   # The v1.1 baseline EMD (0.12048789) is computed on log-returns directly;
   # samples_C are in [-1,1] post-Lambert space, so we need the inverse chain
   # to reconstruct log-returns. Use the existing full_denorm_pipeline:
   from revision.core.data import full_denorm_pipeline
   trans_norm = d["transformed_norm_log_delta"]
   mu = d["mu"]; sigma = d["sigma"]; delta_const = float(d["delta"])
   pipe_out = full_denorm_pipeline(
       torch.tensor(samples_C).double(),
       trans_norm.double(),
       mu.double() if torch.is_tensor(mu) else torch.tensor(mu, dtype=torch.float64),
       sigma.double() if torch.is_tensor(sigma) else torch.tensor(sigma, dtype=torch.float64),

   rel_drift = abs(emd_C - baseline_emd) / baseline_emd
   parity_gate = {
       "emd_C_smoke": emd_C,
       "emd_v1_1_baseline": baseline_emd,
       "rel_drift": rel_drift,
   ```
